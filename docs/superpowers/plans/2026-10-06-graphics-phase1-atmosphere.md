# Graphics Overhaul Phase 1 — Skies, Lighting, Grading — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give Mars, Verdant, Cryo and Forge real panoramic skies, and give every world its own lighting, fog and color grade.

**Architecture:** A build-time script generates four 2048×1024 equirectangular PNGs into `Resources/`, which the existing loader already looks up by name. A new `Sources/WorldLook.swift` holds a per-stage look table and one `applyLook(_:)` entry point that sets lights, fog, environment and camera post-processing, including a procedurally built 16³ color-grading LUT. `configureWorld` and `applyQuality` route through it.

**Tech Stack:** Swift 5.9, SceneKit, CoreGraphics, ImageIO. No new dependencies.

**Spec:** `docs/superpowers/specs/2026-10-06-graphics-overhaul-design.md`

## Global Constraints

- iOS 16+ / macOS 13+; one shared `Sources/` tree builds both targets.
- No changes to `RunnerSimulation`, collisions, hitboxes, lane spacing (2.6 m) or entity placement.
- No downloaded art; every asset is generated by code in this repo.
- Missing assets must keep falling back to procedural images.
- `WorldCacheAudit` bounds hold: ≤2 destinations, ≤96 pooled nodes.
- Reduced Motion and quality tiers keep working; performance tier = no grain.
- Absolute paths in commands are relative to the repo root.
- `swiftc` and `xcodebuild` need to run outside the Claude sandbox (xcrun cache in `/var/folders`, Xcode log store). `$TMPDIR` differs outside the sandbox, so outputs go to `work/` (gitignored).

## Reference facts (verified)

- SceneKit equirect mapping: camera forward (−Z) shows panorama u ≈ 0.75; +X (right) shows u ≈ 1.0; so u increases to the right. v = 0 is the zenith, v = 0.5 the horizon.
- `SCNCamera.colorGrading.contents`: a 2D image of n slices of a unit color cube in a horizontal row; 16³ → 256×16.
- `SCNCamera` defaults: `contrast` 0, `saturation` 1, `whiteBalanceTemperature` 0, `whiteBalanceTint` 0, `grainIntensity` 0.
- Existing horizon landmarks (angular position from the camera): Mars sun sphere u 0.79 / v 0.457; Cryo planet u 0.713 / v 0.433; Forge reactor u 0.75 / v 0.456; Verdant temple u 0.75.

```swift
// Equirectangular 2048x1024; u = 0.75 faces the camera's forward (-Z), v = 0.5 is the horizon.
import Foundation
import CoreGraphics
import ImageIO
import UniformTypeIdentifiers

typealias V3 = SIMD3<Double>

func hex(_ s: String) -> V3 {
    let v = UInt64(s, radix: 16) ?? 0
    return V3(Double((v >> 16) & 255), Double((v >> 8) & 255), Double(v & 255)) / 255
}
func mix(_ a: V3, _ b: V3, _ t: Double) -> V3 { a + (b - a) * t }
func smoothstep(_ e0: Double, _ e1: Double, _ x: Double) -> Double {
    let t = min(1, max(0, (x - e0) / (e1 - e0))); return t * t * (3 - 2 * t)
}
func luminance(_ c: V3) -> Double { 0.2126 * c.x + 0.7152 * c.y + 0.0722 * c.z }

func hash(_ x: Int, _ y: Int, _ z: Int, _ seed: Int) -> Double {
    var h = UInt64(bitPattern: Int64(x &* 374_761_393 &+ y &* 668_265_263 &+ z &* 1_274_126_177 &+ seed &* 144_665))
    h = (h ^ (h >> 13)) &* 0x5851_F42D_4C95_7F2D
    h ^= h >> 29
    return Double(h & 0xFF_FFFF) / Double(0xFF_FFFF)
}

/// Trilinear value noise. Sampling on a cylinder (cos lon, sin lon, lat) keeps u = 0/1 seamless.
func noise(_ p: V3, _ seed: Int) -> Double {
    let i = p.rounded(.down), f = p - i
    let u = f * f * (3 - 2 * f)
    let x = Int(i.x), y = Int(i.y), z = Int(i.z)
    func h(_ dx: Int, _ dy: Int, _ dz: Int) -> Double { hash(x + dx, y + dy, z + dz, seed) }
    let x00 = h(0, 0, 0) + (h(1, 0, 0) - h(0, 0, 0)) * u.x
    let x10 = h(0, 1, 0) + (h(1, 1, 0) - h(0, 1, 0)) * u.x
    let x01 = h(0, 0, 1) + (h(1, 0, 1) - h(0, 0, 1)) * u.x
    let x11 = h(0, 1, 1) + (h(1, 1, 1) - h(0, 1, 1)) * u.x
    let y0 = x00 + (x10 - x00) * u.y, y1 = x01 + (x11 - x01) * u.y
    return y0 + (y1 - y0) * u.z
}

func fbm(_ p: V3, octaves: Int, seed: Int) -> Double {
    var sum = 0.0, amp = 0.5, q = p, norm = 0.0
    for o in 0..<octaves { sum += noise(q, seed + o * 31) * amp; norm += amp; q *= 2.03; amp *= 0.5 }
    return sum / norm
}

func direction(u: Double, v: Double) -> V3 {
    let lon = u * 2 * .pi, lat = (0.5 - v) * .pi
    return V3(cos(lat) * sin(lon), sin(lat), cos(lat) * cos(lon))
}

func render(_ sky: Sky, width w: Int, height h: Int) -> [UInt8] {
    let zenith = hex(sky.zenith), horizon = hex(sky.horizon), ground = hex(sky.ground)
    let cloudLit = hex(sky.cloudLit), cloudShadow = hex(sky.cloudShadow), sunColor = hex(sky.sunColor)
    let nebulaA = hex(sky.nebulaA), nebulaB = hex(sky.nebulaB), auroras = sky.aurora.map(hex)
    let sunDir = direction(u: sky.sunU, v: sky.sunV)
    // Rows are independent; each thread writes only its own row of one shared buffer.
    let buffer = UnsafeMutableBufferPointer<UInt8>.allocate(capacity: w * h * 4)
    buffer.initialize(repeating: 255)
    defer { buffer.deallocate() }
    DispatchQueue.concurrentPerform(iterations: h) { row in
        for col in 0..<w {
            let u = (Double(col) + 0.5) / Double(w), v = (Double(row) + 0.5) / Double(h)
            let lon = u * 2 * .pi, lat = (0.5 - v) * .pi
            let dir = direction(u: u, v: v)
            let e = sin(lat)
            var c = e >= 0 ? mix(horizon, zenith, pow(e, 0.42)) : mix(horizon, ground, min(1, -e * 5))
            c = mix(c, horizon * 1.15, exp(-abs(e) * 18) * 0.5)
            let cyl = V3(cos(lon), sin(lon), lat)
            let sunDot = min(1, max(-1, simd_dot(dir, sunDir)))
            let sunAngle = acos(sunDot)
            if sky.nebula > 0 {
                let n = fbm(cyl * 2.2 + 5, octaves: 5, seed: sky.seed)
                let ridge = 1 - abs(2 * n - 1)
                let weight = pow(ridge, 3) * sky.nebula * smoothstep(0.05, 0.5, e)
                c += mix(nebulaA, nebulaB, fbm(cyl * 1.3 + 9, octaves: 3, seed: sky.seed + 7)) * weight
            }
            if sky.stars > 0 && e > -0.02 {
                let r = hash(col, row, 7, sky.seed)
                if r > 1 - sky.stars {
                    let b = pow((r - (1 - sky.stars)) / sky.stars, 3) * 0.9 + 0.1
                    c += V3(repeating: b * max(0, 1 - luminance(c) * 3))
                }
            }
            for (i, tint) in auroras.enumerated() {
                let center = 0.25 + Double(i) * 0.12
                let wobble = sin(lon * 3 + Double(i) * 1.7) * 0.05 + (fbm(cyl * 1.5 + Double(i), octaves: 3, seed: sky.seed + 50 + i) - 0.5) * 0.1
                let d = e - (center + wobble)
                let band = exp(-max(0, d) * 14 - max(0, -d) * 60)
                let rays = 0.4 + 0.6 * pow(noise(V3(cos(lon) * 40, sin(lon) * 40, Double(i) * 3), sky.seed + 90), 2)
                c += tint * band * rays * 0.55
            }
            if e > sky.cloudLow - 0.02 && e < sky.cloudHigh {
                let w = smoothstep(sky.cloudLow, sky.cloudLow + 0.05, e) * (1 - smoothstep(sky.cloudHigh - 0.1, sky.cloudHigh, e))
                let s = sky.cloudScale
                let p = V3(cos(lon) * s, sin(lon) * s, lat * s * 3.5)
                let n = fbm(p, octaves: 6, seed: sky.seed + 200)
                let density = smoothstep(sky.coverage, sky.coverage + 0.22, n) * w
                if density > 0 {
                    let detail = fbm(p * 2.1 + 3, octaves: 4, seed: sky.seed + 300)
                    let light = (0.35 + 0.65 * pow((sunDot + 1) / 2, 2)) * (0.5 + 0.5 * detail)
                    let lit = mix(cloudShadow, cloudLit, light) + sunColor * exp(-sunAngle * 4) * 0.6
                    c = mix(c, lit, density * 0.92)
                }
            }
            let core = sky.sunSize > 0 ? smoothstep(sky.sunSize, sky.sunSize * 0.8, sunAngle) : 0
            c += sunColor * (core * 3 + exp(-sunAngle * 7) * sky.glow + exp(-sunAngle * 1.6) * sky.glow * 0.35)
            let dither = (hash(col, row, 3, 1) - 0.5) / 255
            for k in 0..<3 { buffer[(row * w + col) * 4 + k] = UInt8(max(0, min(255, (c[k] + dither) * 255))) }
        }
    }
    return Array(buffer)
}

func write(_ pixels: [UInt8], width w: Int, height h: Int, to url: URL) {
    let data = CFDataCreate(nil, pixels, pixels.count)!
    let image = CGImage(width: w, height: h, bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: w * 4,
                        space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.noneSkipLast.rawValue),
                        provider: CGDataProvider(data: data)!, decode: nil, shouldInterpolate: false, intent: .defaultIntent)!
    let destination = CGImageDestinationCreateWithURL(url as CFURL, UTType.png.identifier as CFString, 1, nil)!
    CGImageDestinationAddImage(destination, image, nil)
    precondition(CGImageDestinationFinalize(destination), "Could not write \(url.path)")
}

/// Self-check: size, a seamless wrap, and a visible (non-black) horizon.
func check(_ pixels: [UInt8], width w: Int, height h: Int, name: String) {
    precondition(pixels.count == w * h * 4, "\(name): wrong size")
    var seam = 0.0
    for row in 0..<h { for k in 0..<3 {
        seam += abs(Double(pixels[row * w * 4 + k]) - Double(pixels[(row * w + w - 1) * 4 + k])) / 255
    } }
    seam /= Double(h * 3)
    let horizonRow = h / 2 - 4
    var horizon = 0.0
    for col in 0..<w { horizon += Double(pixels[(horizonRow * w + col) * 4 + 1]) / 255 }
    horizon /= Double(w)
    precondition(seam < 0.03, "\(name): seam \(seam) too visible")
    precondition(horizon > 0.08, "\(name): horizon too dark (\(horizon))")
    print(name, String(format: "seam=%.4f horizon=%.3f", seam, horizon))
}

let output = URL(fileURLWithPath: CommandLine.arguments.dropFirst().first ?? "Resources")
for sky in skies {

    check(pixels, width: 2048, height: 1024, name: sky.name)
    write(pixels, width: 2048, height: 1024, to: output.appendingPathComponent(sky.name + ".png"))
}
```

```swift
import AppKit
import Metal

/// Verifies the grading LUT layout against SceneKit and that every world applies its look.
@main struct LookTests {
    static func meanAbsDifference(_ a: NSImage, _ b: NSImage) -> Double {
        let ra = NSBitmapImageRep(data: a.tiffRepresentation!)!, rb = NSBitmapImageRep(data: b.tiffRepresentation!)!
        var total = 0.0, n = 0
        for y in stride(from: 0, to: ra.pixelsHigh, by: 3) { for x in stride(from: 0, to: ra.pixelsWide, by: 3) {
            let ca = ra.colorAt(x: x, y: y)!.usingColorSpace(.sRGB)!, cb = rb.colorAt(x: x, y: y)!.usingColorSpace(.sRGB)!
            total += abs(ca.redComponent - cb.redComponent) + abs(ca.greenComponent - cb.greenComponent) + abs(ca.blueComponent - cb.blueComponent)
            n += 3
        } }
        return total / Double(n)
    }

    static func main() {
        let resources = URL(fileURLWithPath: FileManager.default.currentDirectoryPath).appendingPathComponent("Resources")
        let world = CoastalWorld(assetsURL: resources)
        world.applyQuality(.ultra)
        let renderer = SCNRenderer(device: MTLCreateSystemDefaultDevice(), options: nil)
        renderer.scene = world.scene; renderer.pointOfView = world.camera
        var run = RunnerSimulation(); run.start(seed: 3, mode: .odyssey, startStage: 0)
        world.update(simulation: run, character: "milo", board: "tideline", time: 1)
        let size = CGSize(width: 160, height: 260)

        // 1. An identity LUT must leave the frame unchanged; a wrong slice/row order would not.
        world.camera.camera?.colorGrading.contents = nil
        let plain = renderer.snapshot(atTime: 1, with: size, antialiasingMode: .none)
        world.camera.camera?.colorGrading.contents = CoastalWorld.gradingLUT { $0 }
        let identity = renderer.snapshot(atTime: 1, with: size, antialiasingMode: .none)
        let identityDiff = meanAbsDifference(plain, identity)
        precondition(identityDiff < 0.012, "Identity LUT changed the frame by \(identityDiff); slice or row order is wrong")
        // 2. An inverting LUT must change it a lot, proving the LUT is applied at all.
        world.camera.camera?.colorGrading.contents = CoastalWorld.gradingLUT { 1 - $0 }
        let inverted = renderer.snapshot(atTime: 1, with: size, antialiasingMode: .none)
        precondition(meanAbsDifference(plain, inverted) > 0.2, "Inverting LUT had no effect")

        // 3. Every world applies its own table, and off-world skies are real assets.
        for (index, stage) in WorldStage.allCases.enumerated() {
            var stageRun = RunnerSimulation(); stageRun.start(seed: 3, mode: .odyssey, startStage: index)
            world.update(simulation: stageRun, character: "milo", board: "tideline", time: 2)
            let camera = world.camera.camera!, look = stage.look
            precondition(abs(Double(camera.contrast) - look.contrast) < 0.001, "\(stage): contrast not applied")
            precondition(abs(Double(camera.saturation) - look.saturation) < 0.001, "\(stage): saturation not applied")
            precondition(camera.colorGrading.contents != nil, "\(stage): no grading LUT")
```

```swift
extension CoastalWorld {
    /// Single entry point for a world's lighting and post-processing; quality only scales it.
    func applyLook(_ stage: WorldStage) {
        let look = stage.look
        worldSun?.color = color(look.sunColor)
        worldSun?.intensity = CGFloat(look.sunIntensity)
        if let sun = worldSunNode {
            let az = look.sunAzimuth * .pi / 180, el = look.sunElevation * .pi / 180
            let target = SCNVector3(0, 0, -8)
            let distance = 32.0
            sun.position = SCNVector3(CoastScalar(sin(az) * cos(el) * distance),
                                      CoastScalar(sin(el) * distance),
                                      CoastScalar(-8 - cos(az) * cos(el) * distance))
            sun.look(at: target)
        }
        worldAmbient?.color = color(look.ambientColor)
        worldAmbient?.intensity = CGFloat(look.ambientIntensity)
        scene.lightingEnvironment.intensity = CGFloat(look.environmentIntensity)
        scene.fogColor = color(look.fogColor)
        scene.fogStartDistance = CGFloat(look.fogStart)
        scene.fogEndDistance = CGFloat(look.fogEnd)
        scene.fogDensityExponent = CGFloat(look.fogExponent)
        guard let lens = camera.camera else { return }
        let fullPost = qualityLevel > 0
        lens.exposureOffset = CGFloat(look.exposure)
        lens.contrast = CGFloat(look.contrast)
        lens.saturation = CGFloat(look.saturation)
        lens.whiteBalanceTemperature = CGFloat(look.temperature)
        lens.whiteBalanceTint = CGFloat(look.tint)
        lens.bloomIntensity = CGFloat(look.bloomIntensity * (fullPost ? 1 : 0.45))
        lens.bloomThreshold = CGFloat(look.bloomThreshold)
        lens.bloomBlurRadius = fullPost ? 8 : 3
        lens.vignettingIntensity = CGFloat(look.vignette)
        lens.vignettingPower = 0.8
        lens.grainIntensity = fullPost ? CGFloat(look.grain) : 0
        lens.grainScale = 1.2
        let shadow = Self.rgb(color(look.shadowTint)), highlight = Self.rgb(color(look.highlightTint))
        lens.colorGrading.contents = Self.gradingLUT { c in
            let l = simd_dot(c, SIMD3(0.2126, 0.7152, 0.0722))
            // Hue-only offsets so the tints never brighten or darken the frame.
            let shadowShift = shadow - simd_dot(shadow, SIMD3(0.2126, 0.7152, 0.0722))
            let highlightShift = highlight - simd_dot(highlight, SIMD3(0.2126, 0.7152, 0.0722))
            var out = c + shadowShift * pow(1 - l, 2) * look.splitStrength + highlightShift * pow(l, 2) * look.splitStrength
            out = simd_clamp(out, SIMD3(repeating: 0), SIMD3(repeating: 1))
            let s = out * out * (3 - 2 * out)
            return out + (s - out) * look.curve
        }
    }

    static func rgb(_ value: CoastColor) -> SIMD3<Double> {
        var r: CGFloat = 0, g: CGFloat = 0, b: CGFloat = 0, a: CGFloat = 0
        value.getRed(&r, green: &g, blue: &b, alpha: &a)
        return SIMD3(Double(r), Double(g), Double(b))
    }

    /// `size` slices of a size³ cube in one horizontal row (SceneKit's documented layout).
    /// Within a slice: x = red, row = green; slice index = blue.
    static func gradingLUT(size: Int = 16, grade: (SIMD3<Double>) -> SIMD3<Double>) -> CGImage? {
        let width = size * size, height = size
        var pixels = [UInt8](repeating: 255, count: width * height * 4)
        let scale = Double(size - 1)
        for g in 0..<size { for b in 0..<size { for r in 0..<size {
            let out = grade(SIMD3(Double(r) / scale, Double(g) / scale, Double(b) / scale))
            let i = (g * width + b * size + r) * 4
            pixels[i] = UInt8(max(0, min(255, out.x * 255 + 0.5)))
            pixels[i + 1] = UInt8(max(0, min(255, out.y * 255 + 0.5)))
            pixels[i + 2] = UInt8(max(0, min(255, out.z * 255 + 0.5)))
        } } }
        guard let provider = CGDataProvider(data: Data(pixels) as CFData) else { return nil }
        return CGImage(width: width, height: height, bitsPerComponent: 8, bitsPerPixel: 32, bytesPerRow: width * 4,
                       space: CGColorSpace(name: CGColorSpace.sRGB)!, bitmapInfo: CGBitmapInfo(rawValue: CGImageAlphaInfo.noneSkipLast.rawValue),
                       provider: provider, decode: nil, shouldInterpolate: false, intent: .defaultIntent)
    }
}
```
